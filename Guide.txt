The error `invalid input syntax for type uuid: "tool_1785337661770"` occurs because your `development_tools` table defines the `id` column as a **UUID**, but your code is trying to insert a string like `"tool_1785337661770"`.

In Supabase, if you define `id uuid DEFAULT uuid_generate_v4()`, you **must** let PostgreSQL generate the ID automatically. You cannot manually pass a string ID unless you change the column type to `text`.

### ✅ The Fix: Let Supabase Generate the UUID

Update your `development-tools-tab.tsx` to **not** provide an `id` when creating new records. Supabase will auto-generate a valid UUID.

#### 1. Update `development-tools-tab.tsx`

```tsx
// development-tools-tab.tsx

const handleSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  
  if (!formData.projectId || !formData.toolName || !formData.quantity || !formData.unitCost) {
    toast.error('Project, tool name, quantity, and unit cost are required');
    return;
  }

  try {
    // ✅ FIX: Do NOT pass an ID for new records. Let Supabase generate it.
    // Only pass the ID if we are editing an existing record.
    const docId = editingId || undefined; 
    
    const qty = parseFloat(formData.quantity);
    const cost = parseFloat(formData.unitCost);
    
    const data = {
      projectId: formData.projectId,
      toolName: formData.toolName,
      quantity: qty,
      unitCost: cost,
      totalCost: qty * cost,
      receiptUrl: formData.receiptUrl,
      currency: formData.currency,
      orgId: slugifyOrg(user!.organizationName),
    };

    // If editing, pass the ID. If creating, pass undefined so Supabase generates it.
    await createOrUpdateDoc('development_tools', docId, data, user!.id, !!editingId);

    setOpen(false);
    toast.success(editingId ? 'Development tool updated' : 'Development tool added');

  }
};
```

#### 2. Update `sync-service.ts` to Handle Undefined IDs

Your `createOrUpdateDoc` function needs to handle the case where `documentId` is `undefined` (for new records). It should only include the `id` field in the upsert payload if it exists.

```typescript
// sync-service.ts

export async function createOrUpdateDoc(
  collectionName: string,
  documentId: string | undefined, // Allow undefined
  data: any,
  userId: string,
  isUpdate: boolean = false
): Promise<void> {
  const timestamp = new Date().toISOString();
  const orgId = data.orgId ? slugifyOrg(data.orgId) : undefined;
  
  const docData = {
    ...data,
    // ✅ FIX: Only include id if it exists (for updates). 
    // For inserts, let Supabase generate it via DEFAULT uuid_generate_v4().
    ...(documentId ? { id: documentId } : {}),
    ...(orgId ? { orgId } : {}),
    ...(data.projectId ? { projectId: (data.projectId as string).toUpperCase() } : {}),
    ...(isUpdate
      ? { updatedAt: timestamp, updatedBy: userId }
      : { createdAt: timestamp, updatedAt: timestamp, createdBy: userId, updatedBy: userId }),
  };

  // If updating, we must specify onConflict. If inserting, Supabase handles it.
  const upsertOptions = documentId ? { onConflict: 'id' } : {};

  const { error } = await supabase
    .from(collectionName)
    .upsert(docData, upsertOptions);

  if (error) {
    console.error(`[DB] Failed to write to ${collectionName}:`, error.message);
    throw new Error(error.message);
  }

  // Audit log...
}
```
